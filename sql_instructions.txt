CREATE TABLE IF NOT EXISTS public.posts (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    title TEXT NOT NULL,
    description TEXT,
    image_url TEXT,
    youtube_url TEXT,
    youtube_video_id TEXT,
    status TEXT NOT NULL DEFAULT 'draft' CHECK (status IN ('draft', 'published')),
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- Index for querying published posts quickly
CREATE INDEX IF NOT EXISTS idx_posts_status ON public.posts(status);

-- Function to automatically update 'updated_at'
CREATE OR REPLACE FUNCTION update_posts_updated_at_column()
RETURNS TRIGGER AS $$
BEGIN
    NEW.updated_at = NOW();
    RETURN NEW;
END;
$$ language 'plpgsql';

-- Trigger to update 'updated_at' on posts
CREATE TRIGGER update_posts_updated_at
BEFORE UPDATE ON public.posts
FOR EACH ROW EXECUTE FUNCTION update_posts_updated_at_column();

-- Enable Row Level Security
ALTER TABLE public.posts ENABLE ROW LEVEL SECURITY;

-- Policies for 'posts' table
-- 1. Public read access ONLY for published posts
CREATE POLICY "Public read access for published posts"
ON public.posts
FOR SELECT
USING (status = 'published');

-- 2. Admin full access (Select, Insert, Update, Delete)
CREATE POLICY "Admin full access for posts"
ON public.posts
FOR ALL
USING (public.is_admin());

-- Storage Bucket setup
INSERT INTO storage.buckets (id, name, public) 
VALUES ('post-images', 'post-images', true) 
ON CONFLICT (id) DO NOTHING;

-- Storage Policies for 'post-images' bucket
-- 1. Public read access
CREATE POLICY "Public read access for post images"
ON storage.objects
FOR SELECT
USING (bucket_id = 'post-images');

-- 2. Admin full access
CREATE POLICY "Admin write access for post images"
ON storage.objects
FOR ALL
USING (bucket_id = 'post-images' AND public.is_admin());
